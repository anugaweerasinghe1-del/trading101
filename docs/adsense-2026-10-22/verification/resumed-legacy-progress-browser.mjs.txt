import assert from 'node:assert/strict';
import {writeFileSync,mkdirSync} from 'node:fs';
import {openAuditPage,pause} from '/workspace/trading101/docs/adsense-2026-10-22/browser-session.mjs';
const [origin,sha,phase,stem]=process.argv.slice(2);
assert.match(sha,/^[a-f0-9]{40}$/);assert.ok(['before','after'].includes(phase));assert.match(stem,/^[A-Z0-9-]+$/);
const root='/workspace/trading101/docs/adsense-2026-10-22/evidence/',p=await openAuditPage(),rows=[],errors=[],blocked=[];
mkdirSync(root+stem.toLowerCase(),{recursive:true});await p.send('Runtime.enable');await p.send('Accessibility.enable');
await p.send('Fetch.enable',{patterns:[{urlPattern:'https://*.supabase.co/*',requestStage:'Request'}]});
p.onEvent(m=>{if(m.method==='Runtime.exceptionThrown')errors.push(m.params.exceptionDetails.text);if(m.method==='Fetch.requestPaused'){const{requestId,request}=m.params,allowed=['GET','HEAD','OPTIONS'].includes(request.method);if(!allowed)blocked.push({method:request.method,path:new URL(request.url).pathname});p.send(allowed?'Fetch.continueRequest':'Fetch.failRequest',allowed?{requestId}:{requestId,errorReason:'Aborted'}).catch(()=>{});}});
try{
 for(let lesson=1;lesson<=6;lesson++)for(const[width,height]of[[360,800],[390,844],[768,1024],[1280,800],[1920,1080]]){
  await p.send('Emulation.setDeviceMetricsOverride',{width,height,deviceScaleFactor:1,mobile:width<500});
  await p.send('Page.navigate',{url:'about:blank'});await p.wait("location.href==='about:blank'&&document.readyState==='complete'");
  await p.send('Page.navigate',{url:origin+'/learn/'+lesson});await p.wait("!!document.querySelector('main [role=progressbar]')&&!document.querySelector('#prerender-seo')");await pause(350);
  const row=await p.evaluate(`(()=>{const r=document.querySelector('main [role=progressbar]'),i=r.firstElementChild;return{width:innerWidth,label:r.parentElement.innerText,ariaValue:r.getAttribute('aria-valuenow'),ariaName:r.getAttribute('aria-label'),state:r.getAttribute('data-state'),indicatorTransform:getComputedStyle(i).transform,indicatorWidth:i.getBoundingClientRect().width,rootBackground:getComputedStyle(r).backgroundColor,indicatorBackground:getComputedStyle(i).backgroundColor,overflow:document.documentElement.scrollWidth>innerWidth+1};})()`);
  assert.ok(row.label.includes('0% Complete'));assert.equal(row.ariaValue,phase==='before'?null:'0');assert.equal(row.ariaName,phase==='before'?null:'Lesson completion');assert.equal(row.state,phase==='before'?'indeterminate':'loading');assert.equal(row.overflow,false);
  const ax=(await p.send('Accessibility.getFullAXTree')).nodes.find(n=>n.role?.value==='progressbar');assert.ok(ax);assert.equal(ax.name?.value||'',phase==='before'?'':'Lesson completion');if(phase==='after')assert.equal(Number(ax.value?.value),0);
  rows.push({lesson,...row,accessibility:{role:ax.role.value,name:ax.name?.value||'',value:ax.value?.value??null}});
  if([1,6].includes(lesson)&&[360,1280].includes(width)){await p.evaluate("[...document.querySelectorAll('button')].find(b=>b.textContent.trim()==='Reject analytics')?.click();true");await p.screenshot(root+stem.toLowerCase()+'/lesson'+lesson+'-'+width+'.png');}
  console.log('PASS '+phase+' lesson'+lesson+' '+width);
 }
 assert.deepEqual(errors,[]);
 writeFileSync(root+stem+'-BROWSER.json',JSON.stringify({checkedAt:new Date().toISOString(),sha,origin,phase,outcome:'PASS',scope:'Actual rendered six public legacy lesson initial states at five widths and actual Chromium accessibility tree. No Continue, quiz, answer, completion, reset or progress/storage operations; all Supabase POSTs prevented. Orange track is the background; green indicator correctly translated offscreen at zero, so no separate visual-percent defect asserted.',rows,runtimeErrors:errors,preventedProtectedRequests:[...new Map(blocked.map(b=>[b.method+b.path,b])).values()]},null,2)+'\n');
}finally{await p.close();}
