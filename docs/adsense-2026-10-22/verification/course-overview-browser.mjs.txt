 }});

 writeFileSync(root+stem+'-BROWSER.json',JSON.stringify({checkedAt:new Date().toISOString(),sha,origin,outcome:failure?'FAIL':'PASS',rows,runtimeErrors:errors,preventedProtectedRequests:blocked,failure,scope:'Actual deployed four authored course overviews at all five widths; every original static paragraph, full lesson card, badge criterion and destination checked.
