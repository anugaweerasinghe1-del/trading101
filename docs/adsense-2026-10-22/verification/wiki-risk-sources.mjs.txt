import{openAuditPage,pause}from'/workspace/trading101/docs/adsense-2026-10-22/browser-session.mjs';
import{writeFileSync,mkdirSync}from'node:fs';import{createHash}from'node:crypto';
const folder='/workspace/scratch/tradehq-audit/wiki-risk-primary';mkdirSync(folder,{recursive:true});
const p=await openAuditPage('about:blank',9223),rows=[['orders','https://www.investor.gov/introduction-investing/investing-basics/how-stock-markets-work/types-orders'],['dca','https://www.investor.gov/introduction-investing/investing-basics/glossary/dollar-cost-averaging'],['risk','https://www.investor.gov/introduction-investing/investing-basics/what-risk']];
try{for(const[name,url]of rows){await p.send('Page.navigate',{url});await p.wait("document.readyState==='complete'");await pause(1500);const d=await p.evaluate("({title:document.title,url:location.href,text:document.body.innerText})");d.capturedAt=new Date().toISOString();d.renderedTextSHA256=createHash('sha256').update(d.text).digest('hex');writeFileSync(folder+'/'+name+'.json',JSON.stringify(d,null,2)+'\n');console.log(JSON.stringify({name,title:d.title,url:d.url,capturedAt:d.capturedAt,renderedTextSHA256:d.renderedTextSHA256,text:d.text}));}}finally{await p.close();}
